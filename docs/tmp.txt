// define the routing rules for the Gin web server
func setupRouter() *gin.Engine {


	// serve static files from /static
	// call with http://localhost:8080/static/img/visoto_light.svg
	router.StaticFile("/favicon.ico", "./static/img/favicon.svg")
	router.Static("/static", "static")

	// serve homepage - handles both / and /index.html
	homeHandler := func(c *gin.Context) {
		c.Header("Content-Type", "text/html; charset=utf-8")
		if err := templates.Get().ExecuteTemplate(c.Writer, "base.html", nil); err != nil {
			c.String(http.StatusInternalServerError, "Template execution error: %v", err)
			return
		}
	}
	router.GET("/", homeHandler)
	router.GET("/index.html", homeHandler)

	// serve resource page with specified iri, get iri from resource path and query resource
	// Example: GET http://localhost:8070/resource/https://register.ld.admin.ch/zefix/company/38909#
	router.GET("/resource/*path", func(c *gin.Context) {

		// extract iri from path and validate
		iri := strings.TrimPrefix(c.Param("path"), "/")
		if _, err := url.ParseRequestURI(iri); err != nil {
			c.String(http.StatusBadRequest, "incorrect resource iri: %v", iri)
			return
		}

		// send query
		data, err := fetchResourceData(iri)
		if err != nil {
			c.String(http.StatusInternalServerError, "Failed to fetch resource data: %v", err)
			return
		}

		// debug: print data here
		//fmt.Printf("SparqlResponse: %+v\n", data)

		// set content type to html
		c.Header("Content-Type", "text/html; charset=utf-8")

		// render the template resource.html with the SparqlResponse as data
		if err := templates.Get().ExecuteTemplate(c.Writer, "resource.html", data); err != nil {
			c.String(http.StatusInternalServerError, "Template execution error: %v", err)
			return
		}
	})

	// Route for embedded SPARQL queries
	// example: http://localhost:8070/embedded/https://register.ld.admin.ch/zefix/company/38909#
	router.GET("/embedded/*path", func(c *gin.Context) {

		// extract iri from path & validate
		iri := strings.TrimPrefix(c.Param("path"), "/")
		if _, err := url.ParseRequestURI(iri); err != nil {
			c.String(http.StatusBadRequest, "incorrect resource iri: %v", iri)
			return
		}

		// Process template with SPARQL preprocessing
		// cleanedTemplate, data, err := sparqlPreproc.ProcessTemplateFile("templates/pages/embedded.html", iri)
		_, data, err := sparqlPreproc.ProcessTemplateFile("templates/pages/embedded.html", iri)
		if err != nil {
			c.String(http.StatusInternalServerError, "Preprocessing error: %v", err)
			return
		}

		fmt.Println("=== Known Templates ===")
		fmt.Println(templates.Get().DefinedTemplates())


		// Execute embedded.html template with data
		c.Header("Content-Type", "text/html; charset=utf-8")
		if err := templates.Get().ExecuteTemplate(c.Writer, "embedded.html", data); err != nil {
			c.String(http.StatusInternalServerError, "Template execution error: %v", err)
		}


	})

	// Route for embedded SPARQL queries
	// example: http://localhost:8070/embedded/https://register.ld.admin.ch/zefix/company/38909#
	router.GET("/embedded2/*path", func(c *gin.Context) {

		// extract iri from path & validate
		iri := strings.TrimPrefix(c.Param("path"), "/")
		if _, err := url.ParseRequestURI(iri); err != nil {
			c.String(http.StatusBadRequest, "incorrect resource iri: %v", iri)
			return
		}

		// Process template with SPARQL preprocessing
		// cleanedTemplate, data, err := sparqlPreproc.ProcessTemplateFile("templates/pages/embedded.html", iri)
		_, data, err := sparqlPreproc.ProcessTemplateFile("templates/pages/embedded.html", iri)
		if err != nil {
			c.String(http.StatusInternalServerError, "Preprocessing error: %v", err)
			return
		}
		// fmt.Println("=== cleanedTemplate ===")
		// fmt.Println(cleanedTemplate)
		// fmt.Println("=== data ===")
		// fmt.Printf("%+v\n", data)
		// fmt.Println("======")

		// Create a new template set and parse the cleaned content FIRST
		// This ensures pageContent and pageTitle blocks are defined before base.html references them
		// clonedTmpl := template.New("embedded")

		// Parse the cleaned template content first (defines pageContent and pageTitle blocks)
		// clonedTmpl, err = clonedTmpl.Parse(cleanedTemplate)
		// if err != nil {
		// 	c.String(http.StatusInternalServerError, "Template parse error: %v", err)
		// 	return
		// }

		// Now parse all layout templates (base.html will see the pageContent block)
		// clonedTmpl, err = clonedTmpl.ParseGlob("templates/layout/*.html")
		// if err != nil {
		// 	c.String(http.StatusInternalServerError, "Failed to parse layout templates: %v", err)
		// 	return
		// }

		// Print all known templates
		fmt.Println("=== Known Templates ===")
		fmt.Println(templates.Get().DefinedTemplates())
		fmt.Println("======================")

		// Print pageContent template if it exists
		if pageContentTmpl := templates.Get().Lookup("pageContent"); pageContentTmpl != nil {
			fmt.Println("=== pageContent Template ===")
			fmt.Printf("Name: %s\n", pageContentTmpl.Name())
			if pageContentTmpl.Tree != nil {
				fmt.Printf("Tree Root: %v\n", pageContentTmpl.Tree.Root)
			}
			fmt.Println("============================")
		} else {
			fmt.Println("WARNING: pageContent template not found!")
		}

		// Try to execute pageContent directly to debug
		fmt.Println("=== Testing pageContent Execution ===")
		var testBuf strings.Builder
		if err := templates.Get().ExecuteTemplate(&testBuf, "pageContent", data); err != nil {
			fmt.Printf("ERROR executing pageContent: %v\n", err)
		} else {
			fmt.Printf("SUCCESS! pageContent output length: %d\n", testBuf.Len())
			if testBuf.Len() > 0 {
				outputLen := testBuf.Len()
				if outputLen > 200 {
					outputLen = 200
				}
				fmt.Printf("First %d chars: %s\n", outputLen, testBuf.String()[:outputLen])
			}
		}
		fmt.Println("=====================================")

		// Test executing base.html
		fmt.Println("=== Testing embedded.html Execution ===")
		var baseBuf strings.Builder
		if err := templates.Get().ExecuteTemplate(&baseBuf, "embedded.html", data); err != nil {
			fmt.Printf("ERROR executing base.html: %v\n", err)
		} else {
			fmt.Printf("SUCCESS! embedded.html output length: %d\n", baseBuf.Len())
			if baseBuf.Len() > 0 {
				outputLen := baseBuf.Len()
				if outputLen > 300 {
					outputLen = 300
				}
				fmt.Printf("First %d chars: %s\n", outputLen, baseBuf.String()[:outputLen])
			}
		}
		fmt.Println("===================================")

		// Execute base.html template with data (it will use pageContent and pageTitle blocks from embedded.html)
		c.Header("Content-Type", "text/html; charset=utf-8")
		if err := templates.Get().ExecuteTemplate(c.Writer, "embedded.html", data); err != nil {
			c.String(http.StatusInternalServerError, "Template execution error: %v", err)
		}
	})

	return router
}
